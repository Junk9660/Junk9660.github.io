Camera detection

Use of UAlarmManager to send information to the Evaluator to broadcast location to NPC

//Evaluator.cpp
void USTEval_Perception::HandleCameraDetection(AActor* DetectedActor, AActor* DetectingCamera, int AreaId)
{
	if (Character->TeamId == AreaId)
	{
		if (DetectedActor && !bCameraInvestigation)
		{
			if (!AIController.IsValid()) return;
			bCameraInvestigation = true;
			if (AIController->CurrentStateTreeState != EStateTreeEvent::Attack)
			{
				AcquiredTarget = DetectedActor;
				LastKnownLocation = AcquiredTarget->GetActorLocation();
				UE_LOG(LogSTEvalPerception, Warning, TEXT("USTEval_Perception::HandleCameraDetection"));

                AIController->AcquiredTarget = DetectedActor;
				AIController->CurrentStateTreeState = EStateTreeEvent::Investigate;
			}
		}
	}
}
